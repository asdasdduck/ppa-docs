# ───────────────────────────────────────────────────────────────────────────────
#                             BUTT IMPACT PHYSICS
# ───────────────────────────────────────────────────────────────────────────────
#
# PPA's own butt and rear-thigh jiggle. Used when [General] UsePPAImpactPhysics = true.
# Every vaginal/anal clap (the same moment the clap sound plays) kicks the butt cheeks
# and rear thighs, which then bounce, swing, ripple and squash back to rest like springs.
# The receiver's own movement also makes them jiggle, so they stay alive between impacts.
#
# While a scene is running PPA fully owns these bones on the receiver, so any CBPC/SMP
# physics on them is overridden for the duration of the scene.
#
# Displacements are in bone units, not skin units. UBE weights the butt bones at ~0.2 and
# the rear thigh bones at ~0.12, so the visible skin motion is roughly 1/5 (butt) or 1/8
# (thigh) of these values. Other bodies may need different values.
#
# Every bounce, swing, ripple, squash and thigh motion is tuned with a Frequency and a Damping:
#   Frequency (Hz) - how fast it wobbles. Lower = heavier, softer, slower. Higher = firmer, quicker.
#   Damping        - how fast the wobble dies out. 1.0 = settles with no wobble at all.
#                    Lower values keep wobbling longer (0.1 = very jiggly, 0.5 = fairly firm).
#
[ButtImpact]
# Turn PPA butt impact physics on or off for this body.
# Body/race overrides can set this to false to skip bodies that don't have these bones.
Enabled = true

# Skeleton bones driven by this system. Their rest pose is read from the skeleton,
# so only change these if your body uses differently named butt/thigh bones.
LeftButtNode = "NPC L Butt"
RightButtNode = "NPC R Butt"
LeftThighNode = "NPC L RearThigh"
RightThighNode = "NPC R RearThigh"

# ─── IMPACT ───

# Master multiplier on every impact. Turn this up for more jiggle, down for subtlety.
# At 0 impacts do nothing, but movement-driven jiggle (the InertiaScale settings) still works.
# Range: 0.0 - 5.0 | Default: 1.0
Strength = 1.0

# Controls how much thrust speed affects impact strength.
# Higher values mean even slow thrusts produce a noticeable bounce.
# Lower values mean only fast/rough thrusts cause a big bounce.
# Impact strength also scales with how deep the thrust is.
# Range: 0.1 - 10.0 | Default: 2.5
SpeedSensitivity = 2.5

# Extra strength multiplier for anal impacts (vaginal impacts always use 1.0).
# Range: 0.0 - 3.0 | Default: 1.0
AnalMultiplier = 1.0

# Random strength difference between the left and right cheek on each impact.
# At 0 both cheeks move identically (looks stiff/robotic).
# At 0.2 there is a natural subtle difference every stroke.
# Above 0.4 it starts looking unnatural.
# Range: 0.0 - 0.5 | Default: 0.2
Variation = 0.2

# Seconds between the first and second cheek reacting. Which cheek goes first is random
# per impact. A small delay sells the ripple across the butt.
# Range: 0.0 - 0.15 | Default: 0.02
CheekDelay = 0.02

# Seconds after a cheek is hit before its rear thigh reacts,
# so the wave looks like it travels down the leg.
# Range: 0.0 - 0.25 | Default: 0.045
ThighDelay = 0.045

# ─── IMPACT DIRECTION ───
# These three are mixed together to decide which way the cheeks get pushed.
# Only their size relative to each other matters, not the absolute values.

# How much the cheeks are pushed along the direction of the thrust (into the body).
# Range: 0.0 - 2.0 | Default: 0.6
InwardWeight = 0.6

# How much the cheeks are pushed up toward the spine.
# Range: 0.0 - 2.0 | Default: 0.45
UpWeight = 0.45

# How much the cheeks are pushed sideways, away from each other.
# Thighs ignore this and only use the inward/up mix.
# Range: 0.0 - 2.0 | Default: 0.35
SpreadWeight = 0.35

# ─── CHEEK BOUNCE ───
# The main back-and-forth movement of each cheek.

# How far the cheek moves on a full-strength impact (roughly the peak of the bounce).
# This is the main "how much jiggle" setting.
# Range: 0.0 - 40.0 | Default: 15.198
Displacement = 15.198

# How fast the cheek bounces, in Hz. Lower = heavier, slower wobble. Higher = firmer, tighter.
# Range: 0.5 - 15.0 | Default: 3.8
Frequency = 3.8

# How quickly the bounce settles. 1.0 = no overshoot. Lower values wobble longer.
# Range: 0.01 - 1.5 | Default: 0.16
Damping = 0.16

# Soft limit on how far a cheek can move in total. Movement approaching this limit is
# smoothly compressed rather than cut off, which prevents stretched/torn meshes from
# stacked impacts. Keep it at or above Displacement or big hits will feel flattened.
# Range: 0.0 - 60.0 | Default: 16.0
MaxDisplacement = 16.0

# Links the two cheeks so movement sloshes from one into the other.
# 0 = each cheek moves independently. Higher = they pull each other into sync.
# Range: 0.0 - 1.0 | Default: 0.25
CheekCoupling = 0.25

# How strongly the receiver's own body movement (the animation moving the hips around)
# makes the cheeks lag behind and jiggle, even without impacts.
# 0 = cheeks only move on impacts.
# Range: 0.0 - 5.0 | Default: 1.0
InertiaScale = 1.0

# ─── CHEEK SWING ───
# The butt bone sits above the bulk of the cheek, so rotating it swings the lower
# cheek like a pendulum. This adds a heavy "hanging" sway on top of the bounce.

# How far the cheek swings on a full-strength impact, in degrees.
# Range: 0 - 90 | Default: 26.0
SwingAngle = 26.0

# How fast the cheek swings back and forth, in Hz. Lower = slower, heavier sway.
# Range: 0.5 - 15.0 | Default: 2.585
SwingFrequency = 2.585

# How quickly the swing settles. 1.0 = no overshoot. Lower values sway longer.
# Range: 0.01 - 1.5 | Default: 0.101
SwingDamping = 0.101

# Soft limit on the total swing angle, in degrees. Swings approaching this are
# smoothly compressed rather than cut off.
# Range: 0 - 90 | Default: 45.0
MaxSwingAngle = 45.0

# How strongly the receiver's own body movement makes the cheeks swing, even without impacts.
# 0 = cheeks only swing on impacts.
# Range: 0.0 - 3.0 | Default: 0.35
SwingInertiaScale = 0.35

# Where the "weight" of the left cheek sits relative to its butt bone. The swing pivots
# around the bone and this point is what hangs from it. X is mirrored for the right cheek.
# Don't change this unless you're tuning for a body with a very different butt shape.
# Default: { x = -2.3, y = 0.0, z = -4.8 }
MassOffset = { x = -2.3, y = 0.0, z = -4.8 }

# ─── RIPPLE & SQUASH ───

# Size of a small, faster jiggle layered over the main bounce. Gives the flesh
# a secondary "ripple" instead of moving as one solid piece.
# Range: 0.0 - 20.0 | Default: 3.0
RippleDisplacement = 3.0

# How fast the ripple jiggles, in Hz. Usually higher than the main Frequency.
# Range: 1.0 - 25.0 | Default: 4.103
RippleFrequency = 4.103

# How quickly the ripple fades. Lower values ripple longer.
# Range: 0.01 - 1.5 | Default: 0.12
RippleDamping = 0.12

# How much the cheeks squash (shrink) on impact before springing back and briefly
# overshooting. Fraction of the bone's scale, so 0.07 = squashes about 7%.
# Range: 0.0 - 0.6 | Default: 0.07
Squash = 0.07

# How fast the squash springs back, in Hz.
# Range: 0.5 - 20.0 | Default: 5.0
SquashFrequency = 5.0

# How quickly the squash settles. Lower values pulse longer.
# Range: 0.01 - 1.5 | Default: 0.3
SquashDamping = 0.3

# ─── REAR THIGH ───
# The upper back of the thighs bounce too, slightly after the cheeks (see ThighDelay).

# How far the rear thigh moves on a full-strength impact.
# Range: 0.0 - 30.0 | Default: 20.739
ThighDisplacement = 20.739

# How fast the thigh bounces, in Hz. Lower = heavier, slower wobble.
# Range: 0.5 - 20.0 | Default: 3.896
ThighFrequency = 3.896

# How quickly the thigh bounce settles. 1.0 = no overshoot. Lower values wobble longer.
# Range: 0.01 - 1.5 | Default: 0.2
ThighDamping = 0.2

# Soft limit on how far a rear thigh can move in total. Movement approaching this is
# smoothly compressed rather than cut off. When it is lower than ThighDisplacement,
# hard hits all top out at about this distance.
# Range: 0.0 - 40.0 | Default: 10.0
ThighMaxDisplacement = 10.0

# How strongly the receiver's own body movement makes the rear thighs jiggle, even without impacts.
# 0 = thighs only move on impacts.
# Range: 0.0 - 5.0 | Default: 0.5
ThighInertiaScale = 0.5

Voice Packs

In ppa-voice-configs, use Speed instead of OnThrust. It follows movement speed, including deep, short strokes. To keep the old timing, replace the trigger’s type and add:

[[VoicePreset.Triggers]]
Type = "Speed"
SpeedRange = [5.0, 30.0]
Gap = [0.5, 0.15]

Old OnThrust entries convert automatically with those defaults. Remove VelocitySmoothFactor and Thrust* from voice [Settings]; they are no longer used. MasterVolume still works.

PenetratedSites = ["Hands"] now matches either hand, and PenetratingSites checks where the actor’s own shaft is inserted. VoiceTypes matches NPCs only; use BaseIDs = ["Skyrim.esm|00000007"] for a player preset.

See Voice Settings for timing and matching examples, or Language to translate the menus.
