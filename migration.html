# ───────────────────────────────────────────────────────────────────────────────
#                             BUTT IMPACT PHYSICS
# ───────────────────────────────────────────────────────────────────────────────
#
# PPA's own butt and rear-thigh jiggle. Used when [General] UsePPAImpactPhysics = true.
# Every vaginal/anal clap (the same moment the clap sound plays) kicks the butt cheeks
# and rear thighs, which then bounce, swing, ripple and squash back to rest like springs.
# The receiver's own movement also makes them jiggle, so they stay alive between impacts.
#
# While a scene is running PPA fully owns these bones on the receiver, so any CBPC/SMP
# physics on them is overridden for the duration of the scene.
#
# Displacements are in bone units, not skin units. UBE weights the butt bones at ~0.2 and
# the rear thigh bones at ~0.12, so the visible skin motion is roughly 1/5 (butt) or 1/8
# (thigh) of these values. Other bodies may need different values.
#
# Every bounce, swing, ripple, squash and thigh motion is tuned with a Frequency and a Damping:
#   Frequency (Hz) - how fast it wobbles. Lower = heavier, softer, slower. Higher = firmer, quicker.
#   Damping        - how fast the wobble dies out. 1.0 = settles with no wobble at all.
#                    Lower values keep wobbling longer (0.1 = very jiggly, 0.5 = fairly firm).
#
[ButtImpact]
# Turn PPA butt impact physics on or off for this body.
# Body/race overrides can set this to false to skip bodies that don't have these bones.
Enabled = true

# Skeleton bones driven by this system. Their rest pose is read from the skeleton,
# so only change these if your body uses differently named butt/thigh bones.
LeftButtNode = "NPC L Butt"
RightButtNode = "NPC R Butt"
LeftThighNode = "NPC L RearThigh"
RightThighNode = "NPC R RearThigh"

# ─── IMPACT ───

# Master multiplier on every impact. Turn this up for more jiggle, down for subtlety.
# At 0 impacts do nothing, but movement-driven jiggle (the InertiaScale settings) still works.
# Range: 0.0 - 5.0 | Default: 1.0

# Controls how much thrust speed affects impact strength.
# Higher values mean even slow thrusts produce a noticeable bounce.
# Lower values mean only fast/rough thrusts cause a big bounce.
# Impact strength also scales with how deep the thrust is.
# Range: 0.1 - 10.0 | Default: 2.5
SpeedSensitivity = 2.5

# Extra strength multiplier for anal impacts (vaginal impacts always use 1.0).
# Range: 0.0 - 3.0 | Default: 1.0
AnalMultiplier = 1.0

# Random strength difference between the left and right cheek on each impact.
# At 0 both cheeks move identically (looks stiff/robotic).
# At 0.2 there is a natural subtle difference every stroke.
# Above 0.4 it starts looking unnatural.
# Range: 0.0 - 0.5 | Default: 0.2

Voice Packs

In ppa-voice-configs, use Speed instead of OnThrust. It follows movement speed, including deep, short strokes. To keep the old timing, replace the trigger’s type and add:

[[VoicePreset.Triggers]]
Type = "Speed"
SpeedRange = [5.0, 30.0]
Gap = [0.5, 0.15]

Old OnThrust entries convert automatically with those defaults. Remove VelocitySmoothFactor and Thrust* from voice [Settings]; they are no longer used. MasterVolume still works.

PenetratedSites = ["Hands"] now matches either hand, and PenetratingSites checks where the actor’s own shaft is inserted. VoiceTypes matches NPCs only; use BaseIDs = ["Skyrim.esm|00000007"] for a player preset.

See Voice Settings for timing and matching examples, or Language to translate the menus.
